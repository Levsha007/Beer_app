ВАЖНОЕ ЗАМЕЧАНИЕ О РАБОТЕ СППР

ВЕРДИКТ (КАЧЕСТВЕННЫЙ/БРАК) НЕ ЗАВИСИТ ОТ ТЕКУЩЕГО ЗНАЧЕНИЯ DELTA X!

DELTA X ИСПОЛЬЗУЕТСЯ ТОЛЬКО ДЛЯ РАСЧЕТА ГРАДАЦИЙ N, НО ИТОГОВЫЙ ВЕРДИКТ ОПРЕДЕЛЯЕТСЯ ПОРОГОВЫМ ПРАВИЛОМ P ≤ 0.5 С ФИКСИРОВАННЫМ DELTA X = 1.0.

  • БАЗОВЫЙ ВЕРДИКТ ВЫЧИСЛЯЕТСЯ ОДИН РАЗ С ФИКСИРОВАННЫМ DELTA X = 1.0
  • ТЕКУЩИЙ DELTA X ВЛИЯЕТ ТОЛЬКО НА ОТОБРАЖЕНИЕ ГРАДАЦИЙ (N), LOG₂(N), CO, GO И ТЕКУЩЕЕ P
  • КОЛИЧЕСТВО КАЧЕСТВЕННЫХ И БРАКОВАННЫХ ТОВАРОВ ОСТАЕТСЯ ПОСТОЯННЫМ ПРИ ЛЮБОМ DELTA X

1. РАСЧЕТ ГРАДАЦИЙ

ФОРМУЛА ИЗ МЕТОДИЧКИ:

N = { 2, ЕСЛИ ЗНАЧЕНИЕ В НОРМЕ
    { (X - XMAX)/Δx + 1, ЕСЛИ X > XMAX
    { (XMIN - X)/Δx + 1, ЕСЛИ X < XMIN

РЕАЛИЗАЦИЯ В КОДЕ (MAIN.PY):

def calculate_gradations(x, xmin, xmax, dx):
    """
    РАСЧЕТ ГРАДАЦИЙ ПО МЕТОДИЧКЕ
    x - РЕАЛЬНОЕ ЗНАЧЕНИЕ ХАРАКТЕРИСТИКИ
    xmin - МИНИМАЛЬНАЯ ГРАНИЦА НОРМЫ
    xmax - МАКСИМАЛЬНАЯ ГРАНИЦА НОРМЫ
    dx - ШАГ ИЗМЕРЕНИЯ (ЧУВСТВИТЕЛЬНОСТЬ)
    """
    if xmin <= x <= xmax:
        return 2  # ЗНАЧЕНИЕ В НОРМЕ
    elif x > xmax:
        # ОТКЛОНЕНИЕ ВВЕРХ
        diff = x - xmax
        n = math.ceil(diff / dx) + 1
        return max(2, min(n, 100))
    else:  # x < xmin
        # ОТКЛОНЕНИЕ ВНИЗ
        diff = xmin - x
        n = math.ceil(diff / dx) + 1
        return max(2, min(n, 100))

2. ПОКАЗАТЕЛИ CH, CO, GO

CH

CH = N

СИГНАЛ НОРМЫ
(КОЛИЧЕСТВО ХАРАКТЕРИСТИК)

CO

CO = Σ LOG₂(Nᵢ)

СИГНАЛ ОТКЛОНЕНИЯ
(СУММА ДВОИЧНЫХ ЛОГАРИФМОВ)

GO

GO = CO / CH

ОТНОШЕНИЕ ОТКЛОНЕНИЕ/НОРМА

РЕАЛИЗАЦИЯ В КОДЕ (MAIN.PY):

# РАСЧЕТ ДЛЯ КОНКРЕТНОГО ПРОДУКТА
chars = db.execute_query(chars_query, (product_id, supplier_id))

current_sum_log2 = 0
n = len(chars)  # CH = N

for ch in chars:
    g = calculate_gradations(
        ch['real_value'],
        ch['min_norm'],
        ch['max_norm'],
        delta_x
    )
    current_sum_log2 += math.log2(g)

# CO = СУММА LOG₂(N)
Co = current_sum_log2
# CH = КОЛИЧЕСТВО ХАРАКТЕРИСТИК
Ch = n
# GO = CO / CH
Go = Co / Ch if Ch > 0 else 0

3. ВЕРОЯТНОСТЬ ПРАВИЛЬНОЙ КЛАССИФИКАЦИИ

P = E-LN2 / GO²

ГДЕ LN2 ≈ 0.693147

РЕАЛИЗАЦИЯ В КОДЕ (MAIN.PY):

import math

if Go > 0:
    P = math.exp(-math.log(2) / (Go * Go))
else:
    P = math.exp(-math.log(2) / 0.0001)

# ОПРЕДЕЛЕНИЕ ВЕРДИКТА
is_quality = P <= 0.5  # P ОТ 0 ДО 0.5 → КАЧЕСТВЕННЫЙ

@app.get("/api/spzr/analyze-all")
async def analyze_all_quality(delta_x: float = 1.0):
    # ПОЛУЧАЕМ ВСЕ КОМБИНАЦИИ ПРОДУКТ-ПОСТАВЩИК
    combinations = db.execute_query(query)
    
    results = []
    for combo in combinations:
        chars = db.execute_query(chars_query, (combo['product_id'], combo['supplier_id']))
        
        # БАЗОВЫЙ ВЕРДИКТ (Δx = 1.0)
        base_sum_log2 = 0
        n = len(chars)
        for ch in chars:
            g = calculate_gradations(ch['real_value'], ch['min_norm'], ch['max_norm'], 1.0)
            base_sum_log2 += math.log2(g)
        
        base_Go = base_sum_log2 / n if n > 0 else 0
        base_P = math.exp(-math.log(2) / (base_Go * base_Go))
        is_quality = base_P <= 0.5  # ← ЭТО ОКОНЧАТЕЛЬНЫЙ ВЕРДИКТ
        
        # ТЕКУЩИЙ РАСЧЕТ С ЗАДАННЫМ Δx
        current_sum_log2 = 0
        for ch in chars:
            g = calculate_gradations(ch['real_value'], ch['min_norm'], ch['max_norm'], delta_x)
            current_sum_log2 += math.log2(g)
        
        results.append({
            'product_id': combo['product_id'],
            'product_name': combo['product_name'],
            'supplier_id': combo['supplier_id'],
            'supplier_name': combo['supplier_name'],
            'metrics': { 'Ch': n, 'Co': round(current_sum_log2, 3),
                         'Go': round(current_sum_log2/n, 3), 'is_quality': is_quality }
        })
    return {"success": True, "results": results}

6. МОДЕЛЬ БАЗЫ ДАННЫХ (INIT.SQL)

CREATE TABLE suppliers (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    address TEXT, phone VARCHAR(20),
    contact_person VARCHAR(100), inn VARCHAR(12),
    email VARCHAR(100), created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT, category VARCHAR(50),
    base_price DECIMAL(10,2), created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE characteristics (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE,
    unit VARCHAR(20), delta_x_default FLOAT DEFAULT 0.5,
    weight INTEGER DEFAULT 12, description TEXT
);

CREATE TABLE product_characteristics (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES products(id) ON DELETE CASCADE,
    supplier_id INTEGER REFERENCES suppliers(id) ON DELETE CASCADE,
    characteristic_id INTEGER REFERENCES characteristics(id) ON DELETE CASCADE,
    min_norm FLOAT NOT NULL, max_norm FLOAT NOT NULL,
    real_value FLOAT NOT NULL, measurement_date TIMESTAMP DEFAULT NOW()
);
